**CRITICAL: LockOSThread invariant (Phase 7 Plan 03 fix commit `e8b95da`):**
`systray.SetIcon` / `systray.SetTooltip` / `systray.AddMenuItem` MUST all run on
the goroutine that called `systray.Run`. The existing `SetTrayError` / `SetTrayIdle`
helpers VIOLATE this invariant — they call systray.* from whatever goroutine
invokes them. This worked by luck in Phase 7 because those helpers were called
only from the startup goroutine which happened to not contend. In Phase 9, the
dispatcher (afterDispatch), auth events, and pause changes all fire from DIFFERENT
goroutines — cross-goroutine systray.* calls WILL intermittently corrupt the tray
window's message queue.

Fix: route all tray state changes through a signal channel (`trayRefreshCh`) that
the tray goroutine drains. The tray goroutine calls `refreshTrayVisual()` which
internally calls systray.SetIcon + systray.SetTooltip.

```go
//go:embed assets/tray/tray-has-queue.ico
var trayHasQueueIcon []byte

// trayState captures every input that determines icon + tooltip. A pure
// helper (computeTrayVisual) maps this to the visuals; callers ONLY mutate
// state and signal the tray goroutine — they never touch systray.* directly.
// Why: D-16 tray icon priority + D-17 tooltip format + T-9-10 Win32 HWND
// affinity (systray.* must run on the tray goroutine).
type trayState struct {
    Mode     string
    Paused   bool
    SignedIn bool
    ErrorMsg string
    Count    int
}

// computeTrayVisual is a pure function — testable without a live systray.
// Priority per D-16: error > has-queue > idle.
// Tooltip per D-17: "go-mapi — {segment} — N pending" (error overrides to "go-mapi — <msg>").
func computeTrayVisual(s trayState) (icon []byte, tooltip string) {
    if s.ErrorMsg != "" {
        return trayErrorIcon, "go-mapi — " + s.ErrorMsg
    }
    // Segment (D-17 priority).
    segment := "Manual"
    switch {
    case s.Paused:
        segment = "Paused"
    case !s.SignedIn:
        segment = "Signed out"
    case s.Mode == "auto-draft":
        segment = "Auto-draft"
    }
    tooltip = fmt.Sprintf("go-mapi — %s — %d pending", segment, s.Count)
    // Icon selection (D-16): has-queue only when signed in AND count > 0.
    if s.Count > 0 && s.SignedIn {
        return trayHasQueueIcon, tooltip
    }
    return trayIdleIcon, tooltip
}
```

```go
func (a *App) setLastError(msg string) {
    a.errorMsgMu.Lock()
    changed := a.lastErrorMsg != msg
    a.lastErrorMsg = msg
    a.errorMsgMu.Unlock()
    if changed {
        a.signalTrayRefresh()
    }
}

func (a *App) getLastError() string {
    a.errorMsgMu.Lock()
    defer a.errorMsgMu.Unlock()
    return a.lastErrorMsg
}

    select {
    case a.trayRefreshCh <- struct{}{}:
    default:
    }
}
```

```go
    mShow := systray.AddMenuItem("Show", "Open main window")
    mPause := systray.AddMenuItem("Pause watching", "Silences toasts and auto-draft; queue still collecting")
    systray.AddSeparator()
    mQuit := systray.AddMenuItem("Quit", "Exit go-mapi")

    logInfo("tray ready: menu items registered (Show, Pause watching, Quit)")

    go func() {
        for {
            select {
            case <-mShow.ClickedCh:
                a.showWindow()
            case <-mPause.ClickedCh:
                if a.isPaused() {

                    mPause.SetTitle("Pause watching")

            case <-a.shutdownCtx.Done():

// refreshTrayVisual reads current state and updates systray icon + tooltip.
// Runs ONLY on the tray goroutine (triggered by trayRefreshCh receive in
// the onTrayReady goroutine loop). Any other caller violates LockOSThread.
func (a *App) refreshTrayVisual() {
    state := trayState{
        Mode:     a.getMode(),
        Paused:   a.isPaused(),
        SignedIn: a.auth != nil && a.auth.Status().Authenticated,
        ErrorMsg: a.getLastError(),
        Count:    0,
    }
    if a.watcher != nil {
        state.Count = len(a.watcher.Snapshot())
    }
    icon, tip := computeTrayVisual(state)
    systray.SetIcon(icon)
    systray.SetTooltip(tip)
}
```

package main

import (
    "fmt"
    "net/url"
    "os"
    "path/filepath"

    toast "git.sr.ht/~jackmordaunt/go-toast/v2"
    "github.com/marcfargas/go-mapi/internal/mapi"
)

// initToasts registers the COM activator + activation callback. MUST be called
// before any emitXxxToast call. Safe to call more than once; underlying lib is
// idempotent.
func initToasts(a *App) error {
    exe, err := os.Executable()
    if err != nil {
        return fmt.Errorf("toast: resolve exe path: %w", err)
    }
    icon := toastIconPath(exe) // helper below — resolves absolute icon path
    if err := toast.SetAppData(toast.AppData{
        AppID:         activeAUMID(),
        GUID:          toastActivatorGUID,
        ActivationExe: exe,
        IconPath:      icon,
    }); err != nil {
        return fmt.Errorf("toast: SetAppData: %w", err)
    }
    toast.SetActivationCallback(func(args string, _ []toast.UserData) {
        // COM thread — dispatch to a goroutine. Do NOT do business logic here.
        go a.handleToastAction(args)
    })
    logInfo("toast: initialized (aumid=%s)", activeAUMID())
    return nil
}

// toastIconPath returns the absolute path to the icon file used in toast
// visuals. jackmordaunt/go-toast requires an absolute path.
func toastIconPath(exePath string) string {
    // Icon lives alongside the exe in dev + prod.
    dir := filepath.Dir(exePath)
    return filepath.Join(dir, "go-mapi.ico")
}

// emitArrivalToast fires a toast for a newly-arrived email.
// Suppressed when the main window is visible AND focused (D-11).
// Privacy: Title = sender display name; Body = subject + "📎 N attachment(s)"
// (omitted when N=0). NEVER includes attachment filenames, recipient email,
// or body text (QUAL-03).
func emitArrivalToast(a *App, e mapi.EmailWithId) {
    if a.isVisible() && windowFocused(a) {
        return
    }
    if e.Message == nil {
        return
    }
    title := displayFrom(e.Message)
    body := e.Message.Subject
    if c := len(e.Message.Attachments); c > 0 {

// clearToastForEmail removes the toast(s) associated with a processed email
// from Action Center (NOTIF-05). Called after MarkProcessed / Delete.
func clearToastForEmail(emailID string) {

    // Clear both the arrival toast (tag = emailID) and any per-email error
    // toast (tag = emailID+":err"). The ":success" tag we DON'T clear — leave
    // it for the user to dismiss (it confirms a successful draft).

        }
    }

// displayFrom returns a privacy-safe sender display string. Prefers the
// name over the address; falls back to "(unknown sender)" if both empty.
func displayFrom(msg *mapi.MailMessage) string {
    if msg.From.Name != "" {
        return msg.From.Name
    }
    if msg.From.Address != "" {
        return msg.From.Address
    }
    return "(unknown sender)"
}

// mustExePath caches os.Executable() for the lifetime of the process. Panics
// on first-call failure — the toast subsystem cannot function without it.
var cachedExePath string

func mustExePath() string {
    if cachedExePath == "" {
        exe, err := os.Executable()
        if err != nil {
            panic(fmt.Errorf("toast: os.Executable: %w", err))
        }
        cachedExePath = exe
    }
    return cachedExePath
}

func windowFocused(a *App) bool {
    // Minimal implementation: trust isVisible() AND the Wails runtime's
    // IsMinimised being false. Future refinement: listen for focus events.
    return a.isVisible()
}

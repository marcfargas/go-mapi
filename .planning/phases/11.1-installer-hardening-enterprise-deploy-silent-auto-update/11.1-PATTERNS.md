```go
func swapInPlace(staged, installed string) (oldPath string, err error) {
	oldPath = fmt.Sprintf("%s.old.%d", installed, os.Getpid())
	fromInstalled, _ := windows.UTF16PtrFromString(installed)
	toOld, _ := windows.UTF16PtrFromString(oldPath)
	if err := windows.MoveFileEx(fromInstalled, toOld, windows.MOVEFILE_REPLACE_EXISTING); err != nil {
		return "", fmt.Errorf("rename installed→old: %w", err)
	}
	fromStaged, _ := windows.UTF16PtrFromString(staged)
	toInstalled, _ := windows.UTF16PtrFromString(installed)
	if err := windows.MoveFileEx(fromStaged, toInstalled, 0); err != nil {
		_ = windows.MoveFileEx(toOld, fromInstalled, windows.MOVEFILE_REPLACE_EXISTING)
		return oldPath, fmt.Errorf("rename staged→installed: %w", err)
	}
	return oldPath, nil
}
```

// updatesStagingDir returns %ProgramData%\go-mapi\updates\ — the silent-update
// staging area (download, verify, atomic swap). SYSTEM + Administrators

func updatesStagingDir() string {

		return dir

		return filepath.Join(pd, "go-mapi", "updates")
	}
	return filepath.Join(os.TempDir(), "go-mapi", "updates")
}

### `src/installer/tasks/go-mapi-auto-update.xml` (NEW — Task Scheduler XML)

**Analog:** **none in repo.** No project-shipped XML files exist (verified via `Glob **/*.xml` — only `node_modules` matches). **Primary source: Microsoft Task Scheduler XML schema** — `https://learn.microsoft.com/en-us/windows/win32/taskschd/task-scheduler-schema`.

**Reference (RESEARCH §Pattern 3 — verbatim):**

```xml
<?xml version="1.0" encoding="UTF-16"?>
<Task version="1.4" xmlns="http://schemas.microsoft.com/windows/2004/02/mit/task">
  <RegistrationInfo>
    <Description>go-mapi silent auto-update — fetches and applies updates without elevating the interactive user.</Description>
    <URI>\go-mapi Auto Update</URI>
  </RegistrationInfo>
  <Triggers>
    <CalendarTrigger>
      <StartBoundary>2026-01-01T03:00:00</StartBoundary>
      <Enabled>true</Enabled>
      <RandomDelay>PT30M</RandomDelay>
      <ScheduleByDay><DaysInterval>1</DaysInterval></ScheduleByDay>
    </CalendarTrigger>
    <BootTrigger>
      <Enabled>true</Enabled>
      <Delay>PT5M</Delay>
    </BootTrigger>
  </Triggers>
  <Principals>
    <Principal id="Author">
      <UserId>S-1-5-18</UserId>
      <RunLevel>HighestAvailable</RunLevel>
    </Principal>
  </Principals>
  <Settings>
    <MultipleInstancesPolicy>IgnoreNew</MultipleInstancesPolicy>
    <DisallowStartIfOnBatteries>false</DisallowStartIfOnBatteries>
    <StopIfGoingOnBatteries>false</StopIfGoingOnBatteries>
    <AllowHardTerminate>true</AllowHardTerminate>
    <StartWhenAvailable>true</StartWhenAvailable>
    <RunOnlyIfNetworkAvailable>true</RunOnlyIfNetworkAvailable>
    <Enabled>true</Enabled>
    <ExecutionTimeLimit>PT12H</ExecutionTimeLimit>
  </Settings>
  <Actions Context="Author">
    <Exec>
      <Command>"INSTDIR_PLACEHOLDER\go-mapi.exe"</Command>
      <Arguments>--update-check-silent</Arguments>
    </Exec>
  </Actions>
</Task>
```

**Encoding constraint:** UTF-16 (per `<?xml version="1.0" encoding="UTF-16"?>`). When committing, write as UTF-16 LE with BOM — this is what `schtasks /create /XML` expects on Win 10/11 (RESEARCH A7).

**Substitution mechanism:** `INSTDIR_PLACEHOLDER` is replaced by NSIS at install time via `nsExec::ExecToLog` calling `powershell.exe -Command "(Get-Content ... -Raw) -replace 'INSTDIR_PLACEHOLDER', [regex]::Escape('$INSTDIR') | Set-Content ..."` (RESEARCH §Pattern 3). Use `powershell.exe`, NOT `pwsh` (Phase 10 specifics — PS 5.1 lowest common denominator).

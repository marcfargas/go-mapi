//go:build windows

package main

import (
    "fmt"
    "syscall"
    "unsafe"

    "golang.org/x/sys/windows"
)

// mutexName is scoped to the Windows session (Local\ prefix), so each RDS user gets one instance.
// Ref: PITFALLS §7, RESEARCH §Pattern 4. Global\ is explicitly rejected.
const mutexName = `Local\go-mapi-singleton-v3`

// windowTitle must match options.App{Title: "go-mapi"} from Plan 02's main.go.
const windowTitle = "go-mapi"

var (
    user32            = syscall.NewLazyDLL("user32.dll")
    procFindWindowW   = user32.NewProc("FindWindowW")
    procSetForeground = user32.NewProc("SetForegroundWindow")
    procShowWindow    = user32.NewProc("ShowWindow")
)

const swRestore = 9

var mutexHandle windows.Handle

// acquireSingleInstance tries to grab the per-session mutex.
// Returns raised=true if another instance already owns it.
func acquireSingleInstance() (raised bool, err error) {
    name, err := windows.UTF16PtrFromString(mutexName)
    if err != nil {
        return false, fmt.Errorf("utf16 mutex name: %w", err)
    }
    h, err := windows.CreateMutex(nil, false, name)
    if err != nil && err != windows.ERROR_ALREADY_EXISTS {
        return false, fmt.Errorf("createmutex: %w", err)
    }
    // When CreateMutex succeeds but the object already existed, err is ERROR_ALREADY_EXISTS AND h is valid.
    // Close the handle we just got (it's a duplicate) and signal raise.
    if err == windows.ERROR_ALREADY_EXISTS {
        if h != 0 {
            _ = windows.CloseHandle(h)
        }
        raiseExistingWindow()
        return true, nil
    }
    mutexHandle = h
    return false, nil
}

func releaseSingleInstance() {
    if mutexHandle != 0 {
        _ = windows.CloseHandle(mutexHandle)
        mutexHandle = 0
    }
}

func raiseExistingWindow() {
    title, _ := windows.UTF16PtrFromString(windowTitle)
    hwnd, _, _ := procFindWindowW.Call(0, uintptr(unsafe.Pointer(title)))
    if hwnd == 0 {
        logInfo("raise: no existing window found (title=%q)", windowTitle)
        return
    }
    procShowWindow.Call(hwnd, uintptr(swRestore))
    procSetForeground.Call(hwnd)
}
```

//go:build windows

package main

import (
    "syscall"
    "unsafe"
)

const (
    wmQueryEndSession = 0x0011
    wmEndSession      = 0x0016
    hwndMessage       = ^uintptr(2) // (HWND)(-3) sentinel for message-only window
)

var (
    kernel32          = syscall.NewLazyDLL("kernel32.dll")
    procGetModuleHandle = kernel32.NewProc("GetModuleHandleW")

    procRegisterClass = user32.NewProc("RegisterClassExW")
    procCreateWindow  = user32.NewProc("CreateWindowExW")
    procDefWindowProc = user32.NewProc("DefWindowProcW")
    procDestroyWindow = user32.NewProc("DestroyWindow")
    procGetMessage    = user32.NewProc("GetMessageW")
    procTranslateMsg  = user32.NewProc("TranslateMessage")
    procDispatchMsg   = user32.NewProc("DispatchMessageW")
    procPostQuitMsg   = user32.NewProc("PostQuitMessage")
)

type wndclassex struct {
    cbSize        uint32
    style         uint32
    lpfnWndProc   uintptr
    cbClsExtra    int32
    cbWndExtra    int32
    hInstance     uintptr
    hIcon         uintptr
    hCursor       uintptr
    hbrBackground uintptr
    lpszMenuName  *uint16
    lpszClassName *uint16
    hIconSm       uintptr
}

type msg struct {
    hwnd    uintptr
    message uint32
    wParam  uintptr
    lParam  uintptr
    time    uint32
    pt      struct{ X, Y int32 }
}

var sessionEndCallback func()

// registerSessionEndHandler creates a hidden message-only HWND, registers a class with a WndProc
// that handles WM_QUERYENDSESSION + WM_ENDSESSION, and starts a message pump goroutine.
// Returns a cancel() that destroys the window and exits the pump.
func registerSessionEndHandler(onEnd func()) (cancel func(), err error) {
    sessionEndCallback = onEnd

    className, _ := syscall.UTF16PtrFromString("GoMapiSessionEndWnd")
    hInstance, _, _ := procGetModuleHandle.Call(0)

    wc := wndclassex{
        cbSize:        uint32(unsafe.Sizeof(wndclassex{})),
        lpfnWndProc:   syscall.NewCallback(sessionEndWndProc),
        hInstance:     hInstance,
        lpszClassName: className,
    }
    procRegisterClass.Call(uintptr(unsafe.Pointer(&wc)))

    hwnd, _, _ := procCreateWindow.Call(
        0, uintptr(unsafe.Pointer(className)), 0,
        0, 0, 0, 0, 0,
        hwndMessage, // parent = HWND_MESSAGE
        0, hInstance, 0,
    )
    if hwnd == 0 {
        return nil, syscall.GetLastError()
    }

    done := make(chan struct{})
    go func() {
        defer close(done)
        var m msg
        for {
            ret, _, _ := procGetMessage.Call(uintptr(unsafe.Pointer(&m)), 0, 0, 0)
            if int32(ret) <= 0 {
                return // WM_QUIT or error
            }
            procTranslateMsg.Call(uintptr(unsafe.Pointer(&m)))
            procDispatchMsg.Call(uintptr(unsafe.Pointer(&m)))
        }
    }()

    cancel = func() {
        procDestroyWindow.Call(hwnd)
        procPostQuitMsg.Call(0)
        <-done
    }
    return cancel, nil
}

func sessionEndWndProc(hwnd uintptr, umsg uint32, wParam, lParam uintptr) uintptr {
    switch umsg {
    case wmQueryEndSession:
        if sessionEndCallback != nil {
            sessionEndCallback()
        }
        return 1 // TRUE — agree to end
    case wmEndSession:
        // No-op: callback already fired on QUERY.
        return 0
    }
    ret, _, _ := procDefWindowProc.Call(hwnd, uintptr(umsg), wParam, lParam)
    return ret
}
```

Create `src/app/watcher_bridge.go`:
```go
package main

import (
    "context"

    wruntime "github.com/wailsapp/wails/v2/pkg/runtime"
    "github.com/marcfargas/go-mapi/internal/mapi"
)

// watcherBridge implements mapi.WatcherCallback. It buffers snapshot-change signals in a 1-slot
// channel (replace-on-write) so bursts coalesce into a single EventsEmit call. The dispatcher
// goroutine drains the slot and calls EventsEmit off any watcher-held locks (Pitfall §5).
type watcherBridge struct {
    ctx      context.Context
    emitter  func(name string, data ...interface{})  // injectable for tests
    onError  func(err error)
    pending  chan struct{}
    done     chan struct{}
    getSnap  func() []mapi.EmailWithId  // set to a.watcher.Snapshot after watcher is created
}

func newWatcherBridge(ctx context.Context, onError func(err error)) *watcherBridge {
    b := &watcherBridge{
        ctx:     ctx,
        emitter: func(name string, data ...interface{}) { wruntime.EventsEmit(ctx, name, data...) },
        onError: onError,
        pending: make(chan struct{}, 1),
        done:    make(chan struct{}),
    }
    go b.dispatch()
    return b
}

func (b *watcherBridge) setSnapshotSource(fn func() []mapi.EmailWithId) { b.getSnap = fn }

func (b *watcherBridge) OnQueueChanged(_ []mapi.EmailWithId) {
    // Drop policy: if a signal is already queued, do not block — the dispatcher will read
    // the latest snapshot when it drains (frontend refetches via GetQueue() on event).
    select {
    case b.pending <- struct{}{}:
    default:
    }
}

func (b *watcherBridge) OnError(err error) {
    if b.onError != nil {
        b.onError(err)
    }
}

func (b *watcherBridge) dispatch() {
    for {
        select {
        case <-b.done:
            return
        case <-b.pending:
            // Emit a no-payload event; frontend refetches via GetQueue() to keep payload marshaling
            // off the hot watcher path (Pitfall §5 / Pitfall §1 — smaller event traffic = smaller RAM).
            b.emitter("queue-update")
        }
    }
}

func (b *watcherBridge) Close() { close(b.done) }
```

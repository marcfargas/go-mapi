# Plan 03-04: Pester 5 installer smoke test + CI workflow

## Goal

Add a Pester 5 smoke test that, on a fresh `windows-latest` runner, compiles the installer via `iscc.exe`, runs the silent install, verifies the on-disk and in-registry state matches the INST-02..06 requirements, runs the silent uninstall, and verifies everything is removed. Add the workflow file that runs the smoke test on every push/PR and on demand.

Create `src/installer/tests/installer.Tests.ps1` using Pester 5 syntax:

```powershell
#Requires -Version 7
#Requires -Modules @{ ModuleName='Pester'; ModuleVersion='5.0.0' }

BeforeAll {
    $script:InstallerPath = Join-Path $PSScriptRoot '..\dist\go-mapi-setup.exe' | Resolve-Path -ErrorAction Stop
    $script:InstallDir = Join-Path $env:ProgramFiles 'go-mapi'
    $script:ProgramData = Join-Path $env:ProgramData 'go-mapi'
    $script:ManifestPath = Join-Path $script:ProgramData 'com.gomapi.host.json'
    $script:BackupPath = Join-Path $script:ProgramData 'uninst\previous-mail-client.json'
    $script:UninstExe = Join-Path $script:InstallDir 'unins000.exe'

    $script:BrowserKeys = @(
        'HKLM:\SOFTWARE\Google\Chrome\NativeMessagingHosts\com.gomapi.host'
        'HKLM:\SOFTWARE\Chromium\NativeMessagingHosts\com.gomapi.host'
        'HKLM:\SOFTWARE\Microsoft\Edge\NativeMessagingHosts\com.gomapi.host'
        'HKLM:\SOFTWARE\BraveSoftware\Brave-Browser\NativeMessagingHosts\com.gomapi.host'
        'HKLM:\SOFTWARE\Vivaldi\NativeMessagingHosts\com.gomapi.host'
    )

    $script:MapiHandlerKey = 'HKLM:\SOFTWARE\Clients\Mail\go-mapi'
    $script:MailClientsKey = 'HKLM:\SOFTWARE\Clients\Mail'
}

Describe 'go-mapi installer smoke test' {

    Context 'Silent install' {
        It 'runs the installer silently and exits 0' {
            $logPath = Join-Path $env:RUNNER_TEMP 'install.log'
            $proc = Start-Process -FilePath $script:InstallerPath `
                -ArgumentList '/VERYSILENT','/SUPPRESSMSGBOXES','/NORESTART',"/LOG=$logPath" `
                -Wait -PassThru
            $proc.ExitCode | Should -Be 0
            Test-Path $logPath | Should -BeTrue
        }

        It 'installs go-mapi.dll to Program Files' {
            Test-Path (Join-Path $script:InstallDir 'go-mapi.dll') | Should -BeTrue
        }

        It 'installs go-mapi-host.exe to Program Files' {
            Test-Path (Join-Path $script:InstallDir 'go-mapi-host.exe') | Should -BeTrue
        }

        It 'registers the MAPI handler key' {
            Test-Path $script:MapiHandlerKey | Should -BeTrue
            (Get-ItemProperty -Path $script:MapiHandlerKey -Name 'DLLPath').DLLPath |
                Should -Match 'go-mapi\.dll$'
        }

        It 'sets go-mapi as the default mail client' {
            $default = (Get-ItemProperty -Path $script:MailClientsKey -Name '(Default)').'(Default)'
            $default | Should -Be 'go-mapi'
        }

        It 'writes the shared native-messaging manifest' {
            Test-Path $script:ManifestPath | Should -BeTrue
            $json = Get-Content $script:ManifestPath -Raw | ConvertFrom-Json
            $json.name | Should -Be 'com.gomapi.host'
            $json.type | Should -Be 'stdio'
            $json.path | Should -Match 'go-mapi-host\.exe$'
            $json.allowed_origins.Count | Should -BeGreaterOrEqual 1
        }

        It 'registers all five browser native-messaging trees' {
            foreach ($key in $script:BrowserKeys) {
                Test-Path $key | Should -BeTrue -Because "expected browser registry tree $key"
                $val = (Get-ItemProperty -Path $key -Name '(Default)').'(Default)'
                $val | Should -Be $script:ManifestPath -Because "key $key should point at the shared manifest"
            }
        }

        It 'writes the previous-mail-client backup JSON' {
            # Backup is only written if a previous client existed. On a fresh runner,
            # the default may be empty — in which case the backup should NOT exist.
            # So this test is permissive: if the file exists, it must be valid JSON
            # with the required fields.
            if (Test-Path $script:BackupPath) {
                $json = Get-Content $script:BackupPath -Raw | ConvertFrom-Json
                $json.PSObject.Properties.Name | Should -Contain 'previousClient'
                $json.PSObject.Properties.Name | Should -Contain 'backedUpAt'
            }
        }
    }

    Context 'Silent uninstall' {
        It 'runs the uninstaller silently and exits 0' {
            Test-Path $script:UninstExe | Should -BeTrue -Because 'uninstaller should exist after install'
            $proc = Start-Process -FilePath $script:UninstExe `
                -ArgumentList '/VERYSILENT','/SUPPRESSMSGBOXES','/NORESTART' `
                -Wait -PassThru
            $proc.ExitCode | Should -Be 0
        }

        It 'removes all five browser registry trees' {
            foreach ($key in $script:BrowserKeys) {
                Test-Path $key | Should -BeFalse -Because "browser registry key $key should be gone"
            }
        }

        It 'removes the MAPI handler key' {
            Test-Path $script:MapiHandlerKey | Should -BeFalse
        }

        It 'removes the shared manifest file' {
            Test-Path $script:ManifestPath | Should -BeFalse
        }

        It 'removes the install directory' {
            # Inno Setup may leave the empty dir if any files remain; accept either gone or empty.
            if (Test-Path $script:InstallDir) {
                (Get-ChildItem -Path $script:InstallDir -Force -ErrorAction SilentlyContinue).Count |
                    Should -Be 0 -Because 'install dir should be empty if present'
            }
        }

        It 'removes the backup JSON' {
            Test-Path $script:BackupPath | Should -BeFalse
        }
    }
}
```
</action>

<action>
Create `.github/workflows/installer-smoke.yml`:

```yaml
name: Installer Smoke Test

on:
  push:
    branches:
      - main
      - develop
    paths:
      - 'src/installer/**'
      - 'src/interceptor/**'
      - 'src/native-host/**'
      - '.github/workflows/installer-smoke.yml'
  pull_request:
    paths:
      - 'src/installer/**'
      - 'src/interceptor/**'
      - 'src/native-host/**'
      - '.github/workflows/installer-smoke.yml'
  workflow_dispatch: {}

concurrency:
  group: installer-smoke-${{ github.ref }}
  cancel-in-progress: true

jobs:
  smoke:
    name: Build + install + uninstall round-trip
    runs-on: windows-latest

    steps:
      - uses: actions/checkout@v4

      - name: Install MinGW, Ninja, CMake, Inno Setup
        run: |
          choco install mingw ninja cmake innosetup --no-progress -y
          echo "C:\ProgramData\chocolatey\lib\mingw\tools\install\mingw64\bin" >> $env:GITHUB_PATH

      - name: Build interceptor DLL
        working-directory: src/interceptor
        run: .\build.ps1 -Config Release -Clean

      - name: Setup Go
        uses: actions/setup-go@v5
        with:
          go-version: '1.21'

      - name: Build native host
        working-directory: src/native-host
        shell: pwsh
        run: |
          go mod tidy
          go build -ldflags "-s -w -X main.Version=0.0.0-smoke" -o build/go-mapi-host.exe .

      - name: Compile installer
        shell: pwsh
        run: |
          & "C:\Program Files (x86)\Inno Setup 6\iscc.exe" `
            /DGOMAPIVersion=0.0.0-smoke `
            src\installer\go-mapi.iss
          if (-not (Test-Path 'src\installer\dist\go-mapi-setup.exe')) {
            throw 'Installer compile produced no go-mapi-setup.exe'
          }

      - name: Install Pester 5
        shell: pwsh
        run: |
          # windows-latest ships Pester 5 by default; this step is a belt-and-braces no-op
          # that forces the correct version if a future runner image ships Pester 4.
          Install-Module -Name Pester -MinimumVersion 5.0.0 -Force -SkipPublisherCheck -Scope CurrentUser
          Get-Module -Name Pester -ListAvailable | Select-Object -First 1

      - name: Run installer smoke test
        shell: pwsh
        run: |
          $config = New-PesterConfiguration
          $config.Run.Path = 'src/installer/tests/installer.Tests.ps1'
          $config.Run.Exit = $true
          $config.Output.Verbosity = 'Detailed'
          $config.TestResult.Enabled = $true
          $config.TestResult.OutputPath = 'installer-smoke-results.xml'
          $config.TestResult.OutputFormat = 'NUnitXml'
          Invoke-Pester -Configuration $config

      - name: Upload Pester results
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: installer-smoke-results
          path: installer-smoke-results.xml
          if-no-files-found: warn

      - name: Upload smoke-built installer
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: go-mapi-setup-smoke
          path: src/installer/dist/go-mapi-setup.exe
          if-no-files-found: warn
```
</action>

## Notes

- The smoke test deliberately builds the installer inline (does not depend on `release.yml` or `build.yml` artifacts) so a single PR touching only installer files runs the full round-trip.
- The previous-mail-client backup test is permissive because a fresh CI runner may not have a default Mail client set at all (`(Default)` value may be empty or missing), which is a valid state the installer handles via the D-09 "skip backup if empty or already go-mapi" rule.
